/* =========================================================
   SMART TAILORING BUSINESS MANAGEMENT SYSTEM
   Database: TailoringShopDB
   ========================================================= */


/* =========================================================
   1. CREATE DATABASE
   ========================================================= */

CREATE DATABASE TailoringShopDB;
GO

USE TailoringShopDB;
GO


/* =========================================================
   2. CREATE TABLES
   ========================================================= */

CREATE TABLE Customer (
    customer_id INT PRIMARY KEY,
    name VARCHAR(100),
    address VARCHAR(200),
    phone_no VARCHAR(20)
);
GO


CREATE TABLE Order_status (
    status_id INT PRIMARY KEY,
    status_name VARCHAR(50) NOT NULL
);
GO


CREATE TABLE Employee (
    employee_id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    specialization VARCHAR(100),
    salary DECIMAL(10,2)
);
GO


CREATE TABLE Fabric (
    fabric_id INT PRIMARY KEY,
    fabric_name VARCHAR(100) NOT NULL,
    type VARCHAR(50),
    unit VARCHAR(20),
    price_per_unit DECIMAL(10,2),
    stock_quantity INT
);
GO


CREATE TABLE Design (
    design_id INT PRIMARY KEY,
    design_name VARCHAR(100) NOT NULL,
    description VARCHAR(255),
    category VARCHAR(50)
);
GO


CREATE TABLE Orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    status_id INT,
    order_date DATE,
    total_amount DECIMAL(10,2),

    FOREIGN KEY (customer_id)
        REFERENCES Customer(customer_id),

    FOREIGN KEY (status_id)
        REFERENCES Order_status(status_id)
);
GO


CREATE TABLE Order_item (
    order_item_id INT PRIMARY KEY,
    order_id INT,
    design_id INT,
    fabric_id INT,
    quantity INT,
    price DECIMAL(10,2),

    FOREIGN KEY (order_id)
        REFERENCES Orders(order_id),

    FOREIGN KEY (design_id)
        REFERENCES Design(design_id),

    FOREIGN KEY (fabric_id)
        REFERENCES Fabric(fabric_id)
);
GO


CREATE TABLE Payment (
    payment_id INT PRIMARY KEY,
    order_id INT,
    amount DECIMAL(10,2),
    payment_date DATE,
    payment_method VARCHAR(50),
    payment_status VARCHAR(50),

    FOREIGN KEY (order_id)
        REFERENCES Orders(order_id)
);
GO


CREATE TABLE Delivery (
    delivery_id INT PRIMARY KEY,
    order_id INT,
    delivery_date DATE,
    delivery_man VARCHAR(100),
    address VARCHAR(200),
    status VARCHAR(50),

    FOREIGN KEY (order_id)
        REFERENCES Orders(order_id)
);
GO


CREATE TABLE Measurement (
    measurement_id INT PRIMARY KEY,
    customer_id INT,
    employee_id INT,
    waist DECIMAL(5,2),
    sleeve DECIMAL(5,2),
    length DECIMAL(5,2),
    neck DECIMAL(5,2),
    shoulder DECIMAL(5,2),
    chest DECIMAL(5,2),
    notes VARCHAR(255),

    FOREIGN KEY (customer_id)
        REFERENCES Customer(customer_id),

    FOREIGN KEY (employee_id)
        REFERENCES Employee(employee_id)
);
GO
